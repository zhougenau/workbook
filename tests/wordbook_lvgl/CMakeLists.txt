cmake_minimum_required(VERSION 3.16)
project(wordbook_layout C CXX)
set(CMAKE_C_STANDARD 11)
set(LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_subdirectory("${ROOT}/managed_components/lvgl__lvgl" lvgl EXCLUDE_FROM_ALL)
set(DATA "${ROOT}/build/wordbook/wordbook_data.c")
if(NOT EXISTS "${DATA}")
    set(DATA "${ROOT}/assets/wordbook/sample.c")
endif()
add_executable(test_wordbook_layout ../test_wordbook_layout.c
    "${ROOT}/main/wordbook_state.c" "${ROOT}/assets/fonts/wordbook_font_16.c" "${DATA}")
target_include_directories(test_wordbook_layout PRIVATE "${ROOT}/main")
target_link_libraries(test_wordbook_layout PRIVATE lvgl)
